const BOT_TOKEN = PropertiesService.getScriptProperties().getProperty('BOT_TOKEN'); const ADMIN_CHAT_ID = PropertiesService.getScriptProperties().getProperty('ADMIN_CHAT_ID'); function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data.type === 'diagnostic_request') { sendDiagnosticRequest(data); } return ContentService .createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: String(error) })) .setMimeType(ContentService.MimeType.JSON); } } function sendDiagnosticRequest(data) { if (!BOT_TOKEN) { throw new Error('BOT_TOKEN не найден'); } if (!ADMIN_CHAT_ID) { throw new Error('ADMIN_CHAT_ID не найден'); } let message = ''; // ========================================== // КОМПАНИЯ / ОФИС // ========================================== if (data.clientType === 'company') { message = '🏢 НОВАЯ ЗАЯВКА — ОБСЛУЖИВАНИЕ ОФИСА\n\n' + '👤 Контактное лицо: ' + (data.contactName || 'Не указано') + '\n' + '📞 Телефон: ' + (data.phone || 'Не указан') + '\n' + '📍 Адрес: ' + (data.address || 'Не указан') + '\n\n' + '☕ Кофемашин: ' + (data.coffeeMachines || 'Не указано') + '\n' + '💧 Пурифайеров / кулеров: ' + (data.waterEquipment || 'Не указано') + '\n' + '🫖 Чайников: ' + (data.kettles || 'Не указано') + '\n\n' + '🔧 Как сейчас обслуживают:\n' + (data.currentService || 'Не указано') + '\n\n' + '🎯 Что требуется:\n' + (data.serviceNeed || 'Не указано') + '\n\n' + '💬 Комментарий:\n' + (data.comment || 'Нет') + '\n\n' + buildTelegramContact(data); } // ========================================== // ЧАСТНОЕ ЛИЦО // ========================================== else { message = '🔧 НОВАЯ ЗАЯВКА НА ДИАГНОСТИКУ\n\n' + '👤 Клиент:\n' + buildTelegramContact(data) + '\n\n' + '☕ Марка: ' + (data.brand || 'Не указана') + '\n' + '📋 Модель: ' + (data.model || 'Не указана') + '\n\n' + '⚠️ Симптом: ' + (data.symptom || 'Не указан') + '\n' + '🕐 Когда возникает: ' + (data.timing || 'Не указано') + '\n' + '🔍 Уточнение: ' + (data.detail || 'Не указано') + '\n\n' + '❗ Код ошибки: ' + (data.error || 'Нет / не указан') + '\n\n' + '🛠 Что уже пробовали:\n' + (data.tried || 'Не указано'); } const url = 'https://api.telegram.org/bot' + BOT_TOKEN + '/sendMessage'; const response = UrlFetchApp.fetch(url, { method: 'post', contentType: 'application/json', payload: JSON.stringify({ chat_id: ADMIN_CHAT_ID, text: message, disable_web_page_preview: true }), muteHttpExceptions: true }); const result = JSON.parse(response.getContentText()); if (!result.ok) { throw new Error( response.getContentText() ); } } // ========================================== // ДАННЫЕ TELEGRAM-КЛИЕНТА // ========================================== function buildTelegramContact(data) { let result = ''; if (data.telegramName) { result += 'Имя: ' + data.telegramName + '\n'; } if (data.telegramUsername) { result += 'Telegram: @' + data.telegramUsername + '\n'; } if (data.telegramId) { result += 'Telegram ID: ' + data.telegramId; } if (!result) { result = 'Telegram-контакт не определён'; } return result; } // ========================================== // ПЕРВИЧНАЯ НАСТРОЙКА ADMIN CHAT ID // ========================================== function setupAdmin() { if (!BOT_TOKEN) { throw new Error( 'BOT_TOKEN не найден' ); } const url = 'https://api.telegram.org/bot' + BOT_TOKEN + '/getUpdates'; const response = UrlFetchApp.fetch(url); const data = JSON.parse( response.getContentText() ); if (!data.ok) { throw new Error( 'Telegram API вернул ошибку: ' + response.getContentText() ); } if ( !data.result || data.result.length === 0 ) { throw new Error( 'Бот не получил /start. ' + 'Открой @CoffeeMachineDiagnostBot, ' + 'отправь /start и запусти setupAdmin ещё раз.' ); } let chatId = null; for ( let i = data.result.length - 1; i >= 0; i-- ) { const update = data.result[i]; if ( update.message && update.message.chat ) { chatId = update.message.chat.id; break; } } if (!chatId) { throw new Error( 'Telegram Chat ID не найден' ); } PropertiesService .getScriptProperties() .setProperty( 'ADMIN_CHAT_ID', String(chatId) ); UrlFetchApp.fetch( 'https://api.telegram.org/bot' + BOT_TOKEN + '/sendMessage', { method: 'post', contentType: 'application/json', payload: JSON.stringify({ chat_id: chatId, text: '✅ Заявки подключены!\n\n' + 'Теперь заявки из диагностики ' + 'кофемашины будут автоматически ' + 'приходить сюда.' }) } ); }
